# Serving dependencies - what the Flask app on EC2 actually imports.
# Training-only packages live in requirements-train.txt.
#
# Was UTF-16 encoded, which pip cannot parse, and listed both opencv-python and
# opencv-python-headless (they install the same cv2 module and clobber each
# other). Transitive dependencies have been dropped: pip resolves those.
#
# Ranges rather than exact pins, because the resolution is not tested here.
# On a working box, `pip freeze > requirements.lock.txt` for real reproducibility.

# web
Flask>=3.0,<4

# model + inference
# torch's floor is 2.4, not 2.2: torch 2.2/2.3 are built against numpy 1.x and
# fail at import with "Failed to initialize NumPy: _ARRAY_API not found" if pip
# resolves numpy 2.x alongside them. 2.4 is the first release that accepts both.
torch>=2.4,<3
torchvision>=0.19,<1
numpy>=1.26,<3

# image decoding and the Grad-CAM overlay
pillow>=10.2,<13
opencv-python-headless>=4.10,<6
imageio>=2.34,<3
av>=12,<19               # imageio's pyav plugin - the HEIC fallback in load_image()

# label_encoder.pkl is a pickled sklearn LabelEncoder, so scikit-learn must be
# importable at runtime even though the app never calls it directly. Pin this to
# the version used for training: unpickling across majors can break.
joblib>=1.4,<2
scikit-learn>=1.4,<2

# AWS: Secrets Manager + the Lambda that archives uploads
boto3>=1.34,<2

# YouTube Data API
requests>=2.32,<3

# local .env fallback when Secrets Manager is unreachable
python-dotenv>=1.0,<2

# Vision explanations run on Amazon Bedrock via the Converse API, which boto3
# already provides -- no separate SDK and no API key (the EC2 instance role is
# the credential). Needs bedrock:InvokeModel on the configured model id.
